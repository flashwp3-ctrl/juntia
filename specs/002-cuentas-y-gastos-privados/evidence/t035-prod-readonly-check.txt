--- check.sql ---
select
  (select count(*) from pg_extension where extname = 'pg_graphql') as pg_graphql_installed,
  (select json_agg(json_build_object('fn', p.proname, 'owner', pg_get_userbyid(p.proowner), 'definer', p.prosecdef, 'path', p.proconfig) order by p.proname)
     from pg_proc p join pg_namespace n on n.oid = p.pronamespace
     where n.nspname = 'juntia_private' and p.prosecdef) as private_definers,
  (select json_agg(json_build_object('table', tablename, 'owner', tableowner, 'rls', rowsecurity) order by tablename)
     from pg_tables where schemaname = 'juntia_data') as data_tables,
  (select json_agg(json_build_object('obj', table_name, 'grantee', grantee, 'priv', privilege_type) order by table_name, grantee, privilege_type)
     from information_schema.role_table_grants
     where table_schema in ('public','juntia_data') and grantee in ('anon','authenticated') and privilege_type <> 'SELECT') as non_select_grants,
  (select json_agg(json_build_object('obj', table_name, 'grantee', grantee) order by table_name, grantee)
     from information_schema.role_table_grants
     where table_schema = 'public' and grantee in ('anon','authenticated')) as public_select_grants,
  (select json_agg(json_build_object('fn', p.proname, 'auth', has_function_privilege('authenticated', p.oid, 'EXECUTE'), 'anon', has_function_privilege('anon', p.oid, 'EXECUTE'), 'invoker', not p.prosecdef) order by p.proname)
     from pg_proc p join pg_namespace n on n.oid = p.pronamespace where n.nspname = 'public') as public_functions,
  (select json_agg(c.relname || ' invoker=' || coalesce(array_to_string(c.reloptions, ','), 'none') order by c.relname)
     from pg_class c join pg_namespace n on n.oid = c.relnamespace where n.nspname = 'public' and c.relkind = 'v') as public_views,
  (select json_agg(tgname order by tgname) from pg_trigger t join pg_class c on c.oid = t.tgrelid join pg_namespace n on n.oid = c.relnamespace
     where n.nspname = 'juntia_data' and not t.tgisinternal) as triggers,
  (select count(*) from juntia_data.transactions) as tx_rows;
