const req=require('node:module').createRequire(process.cwd()+'/package.json');const{chromium}=req('playwright');
const out='specs/001-frontend-foundation/evidence-typography/';const routes=['/','/actividad','/metas','/plan','/balance','/cuenta','/ruta-que-no-existe','/interno/primitivos'];
(async()=>{const b=await chromium.launch();for(const theme of ['light','dark'])for(const width of [390,1440])for(const route of routes){const p=await b.newPage({viewport:{width,height:900},colorScheme:theme,reducedMotion:'reduce'});const res=await p.goto((process.env.BASE||'http://localhost:3400')+route);await p.evaluate(()=>document.fonts.ready);if(route.includes('interno'))await p.locator('[data-skeleton-visible]').first().waitFor({state:'attached'});
const fonts=await p.evaluate(()=>[...document.fonts].filter(f=>f.status==='loaded').map(f=>`${f.family} ${f.style} ${f.weight}`).sort());const name=(route==='/'?'inicio':route.slice(1).replaceAll('/','-'))+`-${width}-${theme}.png`;await p.screenshot({path:out+name,fullPage:true});console.log(JSON.stringify({route,width,theme,status:res.status(),capture:name,fonts}));await p.close()}await b.close()})().catch(e=>{console.error(e);process.exitCode=1});
